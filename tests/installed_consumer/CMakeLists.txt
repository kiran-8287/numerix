cmake_minimum_required(VERSION 3.16)
project(NUMERIXInstalledConsumer LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NUMERIX_INSTALL_PREFIX)
    list(PREPEND CMAKE_PREFIX_PATH "${NUMERIX_INSTALL_PREFIX}")
endif()
find_package(NUMERIX 0.2 CONFIG REQUIRED)
add_executable(consumer_c main.c)
add_executable(consumer_cpp main.cpp)
target_link_libraries(consumer_c PRIVATE NUMERIX::numerix)
target_link_libraries(consumer_cpp PRIVATE NUMERIX::numerix)
enable_testing()
add_test(NAME installed_C COMMAND consumer_c)
add_test(NAME installed_Cpp COMMAND consumer_cpp)
